# useOnSelectionChange()

[Source on GitHub](https://github.com/xyflow/xyflow/blob/main/packages/react/src/hooks/useOnSelectionChange.ts)

這個 hook 讓你可以監聽節點和連線選取的變更。如其名稱所示，當節點_或_連線的選取狀態變更時，你提供的回呼函式就會被呼叫。

> [!WARNING]
>
> 你需要記憶化（memoize）傳入的 `onChange` 處理器，否則這個 hook 將無法正常運作。

```jsx
import { useState } from 'react';
import { ReactFlow, useOnSelectionChange } from '@xyflow/react';

function SelectionDisplay() {
  const [selectedNodes, setSelectedNodes] = useState([]);
  const [selectedEdges, setSelectedEdges] = useState([]);

  // 傳入的處理器必須被記憶化，否則這個 hook 將無法正常運作
  const onChange = useCallback(({ nodes, edges }) => {
    setSelectedNodes(nodes.map((node) => node.id));
    setSelectedEdges(edges.map((edge) => edge.id));
  }, []);

  useOnSelectionChange({
    onChange,
  });

  return (
    <div>
      <p>Selected nodes: {selectedNodes.join(', ')}</p>
      <p>Selected edges: {selectedEdges.join(', ')}</p>
    </div>
  );
}
```
